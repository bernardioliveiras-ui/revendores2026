import test from 'node:test';
import assert from 'node:assert/strict';
import mysql from 'mysql2/promise';
import {hashPassword} from '../lib/passwords.ts';

const url=process.env.CASHBACK_INTEGRATION_URL;
test('cashback HTTP + MySQL: attribution, authorization, idempotency and reconciliation', {skip:!url}, async t => {
  assert.match(process.env.DATABASE_NAME||'',/^mix10_.*_test$/,'Run only against a disposable test database');
  const db=await mysql.createConnection({host:process.env.DATABASE_HOST,port:Number(process.env.DATABASE_PORT),user:process.env.DATABASE_USER,password:process.env.DATABASE_PASSWORD,database:process.env.DATABASE_NAME});
  t.after(()=>db.end());
  const now=new Date().toISOString();const password='Cashback-local-test-123!';
  const hash=await hashPassword(password);
  for(const [id,email,role] of [['cbtest-admin','admin@cbtest.example','admin'],['cbtest-r1','r1@cbtest.example','reseller'],['cbtest-r2','r2@cbtest.example','reseller']]) await db.execute('INSERT INTO users (id,email,name,phone,role,password_hash,must_change_password,created_at,updated_at) VALUES (?,?,?,?,?,?,0,?,?)',[id,email,id,'11999999999',role,hash,now,now]);
  const request=async(path,body,headers={},method='POST') => fetch(url+path,{method,headers:{'content-type':'application/json',...headers},...(body!==undefined?{body:JSON.stringify(body)}:{}),redirect:'manual'});
  const login=async email=>{const response=await request('/api/auth/login',{email,password});assert.equal(response.status,200);return response.headers.get('set-cookie').split(';')[0];};
  const admin=await login('admin@cbtest.example');const r1=await login('r1@cbtest.example');const r2=await login('r2@cbtest.example');
  const ref1res=await request('/api/referrals',{customerName:'Indicado primeiro',whatsapp:'11998887777'},{cookie:r1});assert.equal(ref1res.status,200);const ref1=await ref1res.json();
  const ref2=await (await request('/api/referrals',{customerName:'Indicado segundo',whatsapp:'11998886666'},{cookie:r2})).json();
  const token={'x-affiliate-token':process.env.AFFILIATE_WEBHOOK_TOKEN};
  const base={referralCode:ref1.publicCode,externalOrderId:'cbtest-ext-1',buyerEmail:'buyer@cbtest.example',boxes:20,paymentStatus:'paid'};
  const wallet=async cookie=>(await request('/api/cashback',undefined,{cookie},'GET')).json();
  let credit;
  await t.test('rejects absent token, unpaid orders and invalid quantities without credits',async()=>{
    assert.equal((await request('/api/public/affiliate/qualify',base)).status,401);
    for (const change of [{paymentStatus:'pending'},{boxes:0},{boxes:1.5}]) assert.equal((await request('/api/public/affiliate/qualify',{...base,...change},token)).status,400);
    assert.equal((await wallet(r1)).cashbacks.length,0);
  });
  await t.test('valid external order creates a single pending 1500-cent credit and correct buyer link',async()=>{
    const response=await request('/api/public/affiliate/qualify',base,token);assert.equal(response.status,201);
    const body=await response.json();assert.equal(body.cashbackCents,1500);
    const w=await wallet(r1);assert.equal(w.totals.pendingCents,1500);assert.equal(w.totals.qualifiedOrders,1);credit=w.cashbacks[0];assert.equal(credit.buyerEmail,base.buyerEmail);assert.equal(credit.customerName,'Indicado primeiro');assert.equal(credit.qualifyingBoxes,20);
  });
  await t.test('concurrent retries stay one credit; another referral or changed buyer cannot claim order',async()=>{
    const results=await Promise.all(Array.from({length:10},()=>request('/api/public/affiliate/qualify',base,token)));
    assert.ok(results.every(r=>r.status===200));assert.equal((await wallet(r1)).cashbacks.length,1);
    assert.equal((await request('/api/public/affiliate/qualify',{...base,referralCode:ref2.publicCode},token)).status,409);
    assert.equal((await request('/api/public/affiliate/qualify',{...base,buyerEmail:'other@example.com'},token)).status,409);
    assert.equal((await wallet(r2)).cashbacks.length,0);
  });
  await t.test('BKO actions enforce role, approval and a receipt reference; final states cannot reopen',async()=>{
    assert.equal((await request('/api/admin/cashback',{id:credit.id,status:'approved'},{cookie:r1},'PATCH')).status,403);
    assert.equal((await request('/api/admin/cashback',{id:credit.id,status:'paid',payoutReference:'PIX-01'},{cookie:admin},'PATCH')).status,409);
    assert.equal((await request('/api/admin/cashback',{id:credit.id,status:'approved'},{cookie:admin},'PATCH')).status,200);
    assert.equal((await wallet(r1)).totals.approvedCents,1500);
    assert.equal((await request('/api/admin/cashback',{id:credit.id,status:'paid'},{cookie:admin},'PATCH')).status,400);
    assert.equal((await request('/api/admin/cashback',{id:credit.id,status:'paid',payoutReference:'PIX-01'},{cookie:admin},'PATCH')).status,200);
    const c=(await wallet(r1)).cashbacks[0];assert.equal(c.paidBy,'cbtest-admin');assert.equal(c.payoutReference,'PIX-01');assert.equal(c.status,'paid');
    assert.equal((await request('/api/admin/cashback',{id:credit.id,status:'approved'},{cookie:admin},'PATCH')).status,409);
    assert.equal((await request('/api/admin/cashback',{id:'missing',status:'approved'},{cookie:admin},'PATCH')).status,404);
  });
  await t.test('redirect preserves UTMs and records clicks without generating cashback',async()=>{
    const redirect=await request(`/r/${ref1.publicCode}?utm_source=google&utm_campaign=paid-traffic&utm_term=colageno&ref=WRONG`,undefined,{},'GET');assert.equal(redirect.status,302);
    const destination=new URL(redirect.headers.get('location'));assert.equal(destination.searchParams.get('referralCode'),ref1.publicCode);assert.equal(destination.searchParams.get('ref'),ref1.publicCode);assert.equal(destination.searchParams.get('utm_source'),'google');assert.equal(destination.searchParams.get('utm_term'),'colageno');
    const links=await (await request('/api/referrals',undefined,{cookie:r1},'GET')).json();assert.equal(links.referrals[0].clickCount,1);assert.equal((await wallet(r1)).cashbacks.length,1);
  });
  const [refs]=await db.query('SELECT * FROM affiliate_referrals WHERE public_code=?',[ref1.publicCode]);
  const purchase=async(id,paymentId)=>db.execute('INSERT INTO purchases (id,order_number,name,email,phone,cpf_cnpj,postal_code,street,number,district,city,state,kit,boxes,unit_price_cents,total_cents,asaas_payment_id,referral_id,created_at,updated_at) VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)',[id,id,'Comprador local','local@cbtest.example','11999999999','12345678901','06400000','Rua teste','1','Centro','Barueri','SP','kit20',20,7490,149800,paymentId,refs[0].id,now,now]);
  await purchase('cbtest-local-1','cbtest-payment-1');
  const webhook=async(id,event,paymentId='cbtest-payment-1')=>request('/api/billing/webhook',{id,event,payment:{id:paymentId}},{'asaas-access-token':process.env.ASAAS_WEBHOOK_TOKEN});
  await t.test('webhook confirms only paid events; manual approval and duplicate Asaas events never double credit',async()=>{
    assert.equal((await webhook('cbtest-event-created','PAYMENT_CREATED')).status,200);assert.equal((await wallet(r1)).cashbacks.length,1);
    assert.equal((await webhook('cbtest-event-confirmed','PAYMENT_CONFIRMED')).status,200);
    assert.equal((await webhook('cbtest-event-confirmed','PAYMENT_CONFIRMED')).status,200);
    assert.equal((await webhook('cbtest-event-received','PAYMENT_RECEIVED')).status,200);
    assert.equal((await request('/api/admin/orders',{purchaseId:'cbtest-local-1',status:'paid'},{cookie:admin},'PATCH')).status,200);
    const w=await wallet(r1);assert.equal(w.cashbacks.length,2);assert.equal(w.totals.pendingCents,1500);
  });
  await t.test('refund cancels unpaid cashback and late confirmation never reopens it',async()=>{
    assert.equal((await webhook('cbtest-event-refund','PAYMENT_REFUNDED')).status,200);
    assert.equal((await webhook('cbtest-event-late','PAYMENT_RECEIVED')).status,200);
    const w=await wallet(r1);assert.equal(w.totals.pendingCents,0);assert.equal(w.totals.qualifiedOrders,1);assert.equal(w.cashbacks.find(c=>c.purchaseId==='cbtest-local-1').status,'cancelled');
  });
  await t.test('external payments are acknowledged without creating local orders or retry loops',async()=>{
    const response=await webhook('cbtest-external-event','PAYMENT_RECEIVED','external-payment');
    assert.equal(response.status,200);assert.equal((await response.json()).ignored,true);
  });
  await purchase('cbtest-race-order',null);
  await t.test('known local order whose payment id is not saved yet stays retryable',async()=>{
    const response=await request('/api/billing/webhook',{id:'cbtest-local-race',event:'PAYMENT_RECEIVED',payment:{id:'not-yet-saved',externalReference:'cbtest-race-order'}},{'asaas-access-token':process.env.ASAAS_WEBHOOK_TOKEN});
    assert.equal(response.status,503);
    const [rows]=await db.execute('SELECT event_id FROM asaas_events WHERE event_id=?',['cbtest-local-race']);assert.equal(rows.length,0);
  });
  await purchase('cbtest-local-2','cbtest-payment-2');
  await t.test('cashback insertion failure rolls back both purchase status and webhook event',async()=>{
    await db.query("CREATE TRIGGER fail_cbtest_credit BEFORE INSERT ON affiliate_cashbacks FOR EACH ROW BEGIN IF NEW.purchase_id = 'cbtest-local-2' THEN SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='test rollback'; END IF; END");
    assert.equal((await webhook('cbtest-event-rollback','PAYMENT_RECEIVED','cbtest-payment-2')).status,500);
    const [rows]=await db.execute('SELECT payment_status FROM purchases WHERE id=?',['cbtest-local-2']);assert.equal(rows[0].payment_status,'payment_pending');
    const [events]=await db.execute('SELECT event_id FROM asaas_events WHERE event_id=?',['cbtest-event-rollback']);assert.equal(events.length,0);
    await db.query('DROP TRIGGER fail_cbtest_credit');
    assert.equal((await webhook('cbtest-event-rollback','PAYMENT_RECEIVED','cbtest-payment-2')).status,200);
  });
  await t.test('refund after payout preserves paid ledger and flags BKO reconciliation without touching revenue',async()=>{
    const c=(await wallet(r1)).cashbacks.find(c=>c.purchaseId==='cbtest-local-2');
    await request('/api/admin/cashback',{id:c.id,status:'approved'},{cookie:admin},'PATCH');
    await request('/api/admin/cashback',{id:c.id,status:'paid',payoutReference:'PIX-02'},{cookie:admin},'PATCH');
    assert.equal((await webhook('cbtest-after-payout-refund','PAYMENT_REFUNDED','cbtest-payment-2')).status,200);
    const info=await (await request('/api/admin/overview',undefined,{cookie:admin},'GET')).json();
    assert.equal(info.metrics.cashbackReconciliationCount,1);assert.equal(info.metrics.kitRevenueCents,0);assert.equal(info.metrics.grossSalesCents,0);assert.equal(info.cashbacks.find(row=>row.id===c.id).status,'paid');
  });
});
